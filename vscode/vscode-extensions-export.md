# VS Code Extensions Export

## 한 번에 전체 설치하는 방법

아래 명령어를 터미널에 붙여넣으면 전체 설치됩니다.

```bash
code --install-extension 4ops.terraform \
     --install-extension alefragnani.project-manager \
     --install-extension antfu.iconify \
     --install-extension anthropic.claude-code \
     --install-extension codezombiech.gitignore \
     --install-extension dbaeumer.vscode-eslint \
     --install-extension donjayamanne.githistory \
     --install-extension eamodio.gitlens \
     --install-extension ecmel.vscode-html-css \
     --install-extension editorconfig.editorconfig \
     --install-extension esbenp.prettier-vscode \
     --install-extension fabiospampinato.vscode-highlight \
     --install-extension github.vscode-github-actions \
     --install-extension kamikillerto.vscode-colorize \
     --install-extension lukas-tr.materialdesignicons-intellisense \
     --install-extension matijao.vue-nuxt-snippets \
     --install-extension mechatroner.rainbow-csv \
     --install-extension monokai.theme-monokai-pro-vscode \
     --install-extension mquandalle.graphql \
     --install-extension ms-azuretools.vscode-containers \
     --install-extension ms-azuretools.vscode-docker \
     --install-extension ms-ceintl.vscode-language-pack-ko \
     --install-extension ms-python.debugpy \
     --install-extension ms-python.python \
     --install-extension ms-python.vscode-python-envs \
     --install-extension ms-vscode-remote.remote-containers \
     --install-extension ms-vscode-remote.vscode-remote-extensionpack \
     --install-extension oderwat.indent-rainbow \
     --install-extension pranaygp.vscode-css-peek \
     --install-extension prisma.prisma \
     --install-extension redhat.vscode-yaml \
     --install-extension ryu1kn.text-marker \
     --install-extension usernamehw.errorlens \
     --install-extension vscode-icons-team.vscode-icons \
     --install-extension vue.volar \
     --install-extension wayou.vscode-todo-highlight \
     --install-extension xabikos.javascriptsnippets \
     --install-extension ziyasal.vscode-open-in-github
```
